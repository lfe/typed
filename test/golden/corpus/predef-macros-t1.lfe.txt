;; Golden corpus: static predef macros — TIER 1 ONLY (no macro USE)
;; Tests: c*r, comparison aliases, list*, let*, flet*, do, fun, ?,
;;        MODULE, colon-call, defun/defmacro/flet/fletrec/deftype/defspec
;;        LOWERING. Does NOT test macrolet USE (Tier-2).

(defmodule predef-t1-corpus
  (export (car-test 1) (caar-test 1) (cddr-test 1)
          (neq-test 2) (seq-test 2) (sneq-test 2)
          (list*-test 3) (let*-test 1) (flet*-test 1) (do-test 1)
          (fun-ref 0) (recv-test 0) (module-name 0)
          (colon-call 1)
          (flet-test 1) (fletrec-test 1)
          (add 2)))

;; c*r accessors
(defun car-test (x) (car x))
(defun caar-test (x) (caar x))
(defun cddr-test (x) (cddr x))

;; Comparison aliases: !=, ===, !==
(defun neq-test (a b) (!= a b))
(defun seq-test (a b) (=== a b))
(defun sneq-test (a b) (!== a b))

;; list*
(defun list*-test (a b c) (list* a b c))

;; let*
(defun let*-test (x)
  (let* ((a (+ x 1))
         (b (* a 2)))
    b))

;; flet*
(defun flet*-test (x)
  (flet* ((add1 (n) (+ n 1))
          (dbl (n) (* n 2)))
    (dbl (add1 x))))

;; do loop
(defun do-test (n)
  (do ((i 0 (+ i 1))
       (acc 0 (+ acc i)))
      ((== i n) acc)))

;; fun reference
(defun fun-ref () (fun lists:reverse 1))

;; ? (receive with timeout)
(defun recv-test ()
  (? 1000 'timeout))

;; MODULE
(defun module-name () (MODULE))

;; Colon call sugar
(defun colon-call (x) (erlang:atom_to_list x))

;; --- CL def* lowering (Tier-1 definition transforms) ---

;; defmacro -> define-macro
(defmacro my-const () `42)

;; flet -> let-function
(defun flet-test (x)
  (flet ((helper (n) (+ n 1)))
    (helper x)))

;; fletrec -> letrec-function
(defun fletrec-test (n)
  (fletrec ((loop (i acc)
              (if (== i 0) acc (loop (- i 1) (+ acc i)))))
    (loop n 0)))

;; deftype (Erlang type attribute)
(deftype my-type () (union (integer) (atom)))

;; defspec (Erlang spec attribute)
(defspec add ((integer) (integer)) integer)

(defun add (a b) (+ a b))
